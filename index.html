var cityJmaCodes = { nagoya: "230000", // 愛知縣 (名古屋) fukuoka: "400000" // 福岡縣 (福岡) }; function fetchWeather() { try { var citySelect = document.getElementById('citySelect'); if (!citySelect) return; var cityKey = citySelect.value; var code = cityJmaCodes[cityKey]; // 直接抓取日本氣象廳 (JMA) 官方 API var url = 'https://www.jma.go.jp/bosai/forecast/data/forecast/' + code + '.json'; var tempDisplay = document.getElementById('tempDisplay'); var popDisplay = document.getElementById('popDisplay'); if (tempDisplay) tempDisplay.textContent = '讀取...'; if (popDisplay) popDisplay.textContent = '☔ 降雨率 --%'; fetch(url) .then(function(res) { return res.json(); }) .then(function(data) { weatherDataCache = parseJmaData(data); renderWeather(selectedDayIndex); }) .catch(function(err) { console.log(err); }); } catch(e) { console.log(e); } } // 解析日本氣象廳資料格式 function parseJmaData(data) { var result = { temperature_2m_max: [], temperature_2m_min: [], precipitation_probability_max: [], weathercode: [] }; try { var timeSeries = data[0].timeSeries; // 降雨機率 (pop) var pops = timeSeries[1].areas[0].pops; result.precipitation_probability_max = [ Math.max.apply(null, pops.slice(0, 4).map(Number) || [0]), Math.max.apply(null, pops.slice(4, 6).map(Number) || [0]), 0 ]; // 氣溫 (temps) var temps = timeSeries[2].areas[0].temps; result.temperature_2m_min = [temps[0] || "--", temps[2] || "--", "--"]; result.temperature_2m_max = [temps[1] || "--", temps[3] || "--", "--"]; } catch(e) {} return result; }
